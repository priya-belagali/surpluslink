"""Optional language-model assistant: turn a sentence into a pre-filled donation form.

Off unless ANTHROPIC_API_KEY is set. The text typed is sent to the Anthropic API, so do not type anything private.
The model only suggests values. Every suggestion is validated here, and the donor reviews the form before posting.
"""
import json
import os
import re
import urllib.request
from datetime import time

from core import config, safety

API_URL = "https://api.anthropic.com/v1/messages"
CATEGORIES = list(safety.SAFE_HOURS)


class AssistantError(Exception):
    """Anything that stops the assistant from returning usable values. The message is safe to show."""


def available() -> bool:
    return bool(os.getenv("ANTHROPIC_API_KEY"))


def system_prompt() -> str:
    cats = "; ".join(f'"{c}"' for c in CATEGORIES)
    return (
        "You turn a short description of leftover food into fields for a donation form. "
        "Reply with ONE JSON object and nothing else, using exactly these keys: "
        '"title" (short text), "category" (one of: ' + cats + '), "veg" (true or false), '
        '"quantity_meals" (whole number), "cooked_time" (24-hour "HH:MM"), "temperature_c" (number), "notes" (text). '
        "Use null for anything the description does not say. Never guess a number. "
        "If the description says plates, portions or servings, treat each as one meal."
    )


def validate(raw: dict) -> dict:
    """Keep only values of the right type and range. Anything else is dropped (the form keeps its default)."""
    out: dict = {}
    title = raw.get("title")
    if isinstance(title, str) and title.strip():
        out["title"] = title.strip()[:160]
    if raw.get("category") in CATEGORIES:
        out["category"] = raw["category"]
    if isinstance(raw.get("veg"), bool):
        out["veg"] = raw["veg"]
    q = raw.get("quantity_meals")
    if isinstance(q, (int, float)) and not isinstance(q, bool) and 1 <= q <= 100000:
        out["quantity_meals"] = int(q)
    ct = raw.get("cooked_time")
    if isinstance(ct, str):
        m = re.fullmatch(r"\s*([01]?\d|2[0-3]):([0-5]\d)\s*", ct)
        if m:
            out["cooked_time"] = time(int(m.group(1)), int(m.group(2)))
    t = raw.get("temperature_c")
    if isinstance(t, (int, float)) and not isinstance(t, bool) and -30 <= t <= 120:
        out["temperature_c"] = float(t)
    notes = raw.get("notes")
    if isinstance(notes, str) and notes.strip():
        out["notes"] = notes.strip()[:500]
    return out


def parse_response(text: str) -> dict:
    """Pull the JSON object out of the model's reply (it may wrap it in a code fence) and validate it."""
    match = re.search(r"\{.*\}", text or "", re.S)
    if not match:
        raise AssistantError("The assistant did not return usable values. Please fill the form by hand.")
    try:
        raw = json.loads(match.group(0))
    except ValueError:
        raise AssistantError("The assistant's reply could not be read. Please fill the form by hand.") from None
    if not isinstance(raw, dict):
        raise AssistantError("The assistant's reply was not in the expected form.")
    values = validate(raw)
    if not values:
        raise AssistantError("Nothing in that description could be turned into form fields.")
    return values


def parse_listing_text(description: str) -> dict:
    key = os.getenv("ANTHROPIC_API_KEY")
    if not key:
        raise AssistantError("The assistant is not switched on. Set ANTHROPIC_API_KEY to enable it.")
    body = {"model": config.LLM_MODEL, "max_tokens": 400, "system": system_prompt(),
            "messages": [{"role": "user", "content": (description or "")[:1000]}]}
    req = urllib.request.Request(
        API_URL, data=json.dumps(body).encode(),
        headers={"content-type": "application/json", "x-api-key": key, "anthropic-version": "2023-06-01"})
    try:
        with urllib.request.urlopen(req, timeout=25) as resp:
            data = json.load(resp)
    except Exception as exc:
        print(f"[assistant] request failed: {exc}")
        raise AssistantError("The assistant could not be reached. Please fill the form by hand.") from None
    text = "".join(b.get("text", "") for b in data.get("content", []) if b.get("type") == "text")
    return parse_response(text)

"""Registration and login. Passwords are salted PBKDF2 hashes (never stored in plain text)."""
import hashlib
import hmac
import os

from sqlalchemy import select

from core import config
from core.db import User

_ITERATIONS = 200_000


def hash_password(password: str) -> str:
    salt = os.urandom(16)
    digest = hashlib.pbkdf2_hmac("sha256", password.encode(), salt, _ITERATIONS)
    return f"{salt.hex()}${digest.hex()}"


def verify_password(password: str, stored: str) -> bool:
    try:
        salt_hex, hash_hex = stored.split("$")
        salt = bytes.fromhex(salt_hex)
    except ValueError:
        return False
    digest = hashlib.pbkdf2_hmac("sha256", password.encode(), salt, _ITERATIONS)
    return hmac.compare_digest(digest.hex(), hash_hex)


def register_user(session, *, email: str, password: str, name: str, role: str, **extra) -> User:
    email = email.strip().lower()
    if role not in ("donor", "ngo", "volunteer"):
        raise ValueError("Invalid role.")
    if "@" not in email or "." not in email:
        raise ValueError("Please enter a valid email address.")
    if len(password) < 6:
        raise ValueError("Password must be at least 6 characters.")
    if not name.strip():
        raise ValueError("Please enter your name.")
    if session.scalar(select(User).where(User.email == email)):
        raise ValueError("An account with this email already exists.")

    user = User(
        email=email,
        password_hash=hash_password(password),
        name=name.strip(),
        role=role,
        verified=(role != "ngo"),  # NGOs must be approved by an admin before claiming food
        **extra,
    )
    session.add(user)
    session.flush()
    return user


def authenticate(session, email: str, password: str) -> User | None:
    user = session.scalar(select(User).where(User.email == email.strip().lower()))
    if user and verify_password(password, user.password_hash):
        return user
    return None


def ensure_admin(session) -> None:
    email = config.ADMIN_EMAIL.lower()
    if session.scalar(select(User).where(User.email == email)):
        return
    session.add(
        User(
            email=email,
            password_hash=hash_password(config.ADMIN_PASSWORD),
            name="Administrator",
            role="admin",
            verified=True,
        )
    )

@contextmanager
def session_scope():
    """Commit on success, roll back on error, always close."""
    session = SessionLocal()
    try:
        yield session
        session.commit()
    except Exception:
        session.rollback()
        raise
    finally:
        session.close()

_NEW_COLUMNS = {
    "users": [("fssai_number", "VARCHAR(40)"), ("darpan_id", "VARCHAR(40)")],
    "listings": [("temperature_c", "FLOAT"), ("batch_ref", "VARCHAR(40)"), ("qty_distributed", "INTEGER"),

    "offers": [("resolved_at", "TIMESTAMP")],
}


def _ensure_columns() -> None:
    insp = inspect(engine)
    with engine.begin() as conn:
        for table, columns in _NEW_COLUMNS.items():
            if not insp.has_table(table):
                continue
            have = {c["name"] for c in insp.get_columns(table)}
            for name, sql_type in columns:
                if name not in have:
                    conn.execute(text(f"ALTER TABLE {table} ADD COLUMN {name} {sql_type}"))


def init_db() -> None:
    from core.auth import ensure_admin

    Base.metadata.create_all(engine)
    _ensure_columns()
    with session_scope() as session:
        ensure_admin(session)
    if config.SEED_DEMO_ON_EMPTY:
        from core.demo import seed_if_empty

        with session_scope() as session:
            seed_if_empty(session)

"""Surplus forecasting.

With two weeks or more of history: the average of the same weekday in earlier weeks, counting days with no
donation as zero, with a range from how much those days varied. With less history: a rougher average of the
days that had donations. A back-test (see learning.backtest) checks it against simply assuming the overall
average. Candidate models with more parameters (a regression on weekday and trend) were tried on demo-like
data and did not beat this, so they are not used.
"""
from datetime import date

import pandas as pd
from sqlalchemy import select

from core import learning
from core.db import Listing


def forecast_detail(rows: list[tuple], target: date) -> dict:
    """Expected meals for `target` with a low-to-high range.

    Uses the same weekday in past weeks when there are at least two such days, otherwise the
    overall daily average. Returns expected=None when there is too little history.
    """
    empty = {"expected": None, "low": None, "high": None, "days": 0, "basis": None}
    if not rows:
        return empty
    better = learning.weekday_forecast(rows, target)
    if better is not None:
        return better
    df = pd.DataFrame(rows, columns=["created_at", "meals"])
    df["date"] = pd.to_datetime(df["created_at"]).dt.normalize()
    daily = df.groupby("date")["meals"].sum()
    same_weekday = daily[daily.index.dayofweek == target.weekday()]
    if len(same_weekday) >= 2:
        sample, basis = same_weekday, "same weekday"
    elif len(daily) >= 3:
        sample, basis = daily, "all days"
    else:
        return {**empty, "days": len(daily)}
    return {"expected": float(sample.mean()), "low": float(sample.min()), "high": float(sample.max()),
            "days": len(sample), "basis": basis}

from math import asin, cos, radians, sin, sqrt


def haversine_km(lat1: float, lon1: float, lat2: float, lon2: float) -> float:
    """Great-circle distance in kilometres."""
    lat1, lon1, lat2, lon2 = map(radians, (lat1, lon1, lat2, lon2))
    a = sin((lat2 - lat1) / 2) ** 2 + cos(lat1) * cos(lat2) * sin((lon2 - lon1) / 2) ** 2
    return 2 * 6371.0088 * asin(sqrt(a))

    """Turn a Nominatim /search response into {lat, lon, name}, or None."""
    if not data:
        return None
    first = data[0]
    try:
        return {"lat": float(first["lat"]), "lon": float(first["lon"]), "name": str(first.get("display_name", ""))}
    except (KeyError, TypeError, ValueError):
        return None

    try:
        return json.loads(config.GEOCODE_CACHE.read_text(encoding="utf-8"))
    except (OSError, ValueError):
        return {}


def _save_cache(cache: dict) -> None:
    try:
        config.GEOCODE_CACHE.write_text(json.dumps(cache), encoding="utf-8")
    except OSError:
        pass


def geocode(query: str) -> dict | None:
    q = " ".join((query or "").split())
    if len(q) < 4:
        return None
    key = q.lower()
    cache = _load_cache()
    if key in cache:
        return cache[key]

    with _lock:  # keep to one request per second across all visitors
        wait = 1.1 - (time.time() - _last_call[0])
        if wait > 0:
            time.sleep(wait)
        _last_call[0] = time.time()
        params = urllib.parse.urlencode({"q": q, "format": "jsonv2", "limit": 1, "countrycodes": "in"})
        req = urllib.request.Request(f"https://nominatim.openstreetmap.org/search?{params}",
                                     headers={"User-Agent": _user_agent(), "Accept-Language": "en"})
        try:
            with urllib.request.urlopen(req, timeout=8) as resp:
                data = json.load(resp)
        except Exception as exc:
            print(f"[geocode] lookup failed: {exc}")
            return None

    result = parse_result(data)
    if result:
        cache[key] = result
        _save_cache(cache)
    return result

"""Small, explainable learning routines.

Nothing here touches the database or Streamlit, so each routine can be tested and simulated on its own.

1. Offer outcomes -> how likely each NGO is to accept, and how long it typically takes (Bayesian smoothing).
2. Daily surplus -> a ridge regression on weekday and trend, with a walk-forward back-test.
"""
from __future__ import annotations

from dataclasses import dataclass
from datetime import date, timedelta

import numpy as np

# ---------------------------------------------------------------- NGO behaviour

PRIOR_ACCEPT = (2.0, 2.0)  # Beta(2, 2): with no history we assume about 50 percent, and move slowly
MINUTES_PER_KM = 2.0       # rough urban travel speed (30 km/h), to put waiting time and distance on one scale


@dataclass
class NgoHistory:
    accepted: int = 0
    declined: int = 0
    expired: int = 0           # the NGO never responded before the offer ran out
    accept_minutes: float = 0.0  # total time taken by offers that were accepted
    fail_minutes: float = 0.0    # total time consumed by offers that were declined or expired

    @property
    def total(self) -> int:
        return self.accepted + self.declined + self.expired

    @property
    def failed(self) -> int:
        return self.declined + self.expired


def acceptance_probability(accepted: int, total: int, prior: tuple[float, float] = PRIOR_ACCEPT) -> float:
    """Share of offers accepted, pulled towards 50 percent when there is little evidence."""
    a, b = prior
    return (accepted + a) / (total + a + b)


def smoothed_mean(total_minutes: float, count: int, prior_mean: float, prior_weight: float = 2.0) -> float:
    """Average that starts at prior_mean and moves towards the data as observations arrive."""
    return (total_minutes + prior_mean * prior_weight) / (count + prior_weight)


def expected_wait_minutes(h: NgoHistory, timeout_min: float) -> float:
    """Expected minutes before this NGO resolves an offer (accepts it, or fails and we must move on)."""
    p = acceptance_probability(h.accepted, h.total)
    accept_mean = smoothed_mean(h.accept_minutes, h.accepted, prior_mean=timeout_min / 3)
    fail_mean = smoothed_mean(h.fail_minutes, h.failed, prior_mean=timeout_min)
    return p * accept_mean + (1 - p) * fail_mean


def wait_penalty_km(wait_minutes: float) -> float:
    """Waiting time expressed as extra kilometres, so it can be added to the distance score."""
    return wait_minutes / MINUTES_PER_KM


# ---------------------------------------------------------------- surplus forecast

def daily_totals(rows: list[tuple], as_of: date) -> tuple[list[date], np.ndarray]:
    """Meals per calendar day up to as_of. Days with no donation count as zero, which matters:
    leaving them out makes a donor who gives only now and then look like a steady one."""
    totals: dict[date, float] = {}
    for when, meals in rows:
        d = when.date() if hasattr(when, "date") else when
        if d <= as_of:
            totals[d] = totals.get(d, 0.0) + float(meals)
    if not totals:
        return [], np.array([])
    start = min(totals)
    days = [start + timedelta(days=i) for i in range((as_of - start).days + 1)]
    return days, np.array([totals.get(d, 0.0) for d in days])


def weekday_forecast(rows: list[tuple], target: date, as_of: date | None = None, min_days: int = 14) -> dict | None:
    """Expected meals on `target` from the same weekday in earlier weeks, zero days included.

    The range is the mean plus or minus 1.28 standard deviations of those same-weekday totals (about 80
    percent of days if they are roughly bell-shaped). None when there is less than `min_days` of history.
    """
    as_of = as_of or (target - timedelta(days=1))
    days, y = daily_totals(rows, as_of)
    if len(days) < min_days:
        return None
    same = np.array([v for d, v in zip(days, y) if d.weekday() == target.weekday()])
    sample = same if len(same) >= 2 else y
    mean = float(sample.mean())
    sd = float(sample.std(ddof=1)) if len(sample) > 1 else 0.0
    return {"expected": mean, "low": max(mean - 1.28 * sd, 0.0), "high": mean + 1.28 * sd,
            "days": len(sample), "basis": "same weekday, zero days included" if len(same) >= 2 else "all days", "sigma": sd}


def naive_forecast(rows: list[tuple], as_of: date) -> float | None:
    """The comparison point: tomorrow will look like the overall daily average."""
    days, y = daily_totals(rows, as_of)
    return float(y.mean()) if len(days) else None


def backtest(rows: list[tuple], as_of: date, horizon_days: int = 14, min_train: int = 14) -> dict | None:
    """Walk forward over the last `horizon_days`, predicting each day from earlier data only.

    Returns the average absolute error, in meals per day, of the weekday forecast and of the naive average.
    """
    days, actual = daily_totals(rows, as_of)
    if len(days) < min_train + 7:
        return None
    err_weekday, err_naive = [], []
    for i in range(len(days) - horizon_days, len(days)):
        if i < min_train:
            continue
        prior_end = days[i] - timedelta(days=1)
        model = weekday_forecast(rows, days[i], as_of=prior_end, min_days=min_train)
        naive = naive_forecast(rows, prior_end)
        if model is None or naive is None:
            continue
        err_weekday.append(abs(model["expected"] - actual[i]))
        err_naive.append(abs(naive - actual[i]))
    if len(err_weekday) < 5:
        return None
    w, n = float(np.mean(err_weekday)), float(np.mean(err_naive))
    return {"mae_weekday": w, "mae_naive": n, "days_tested": len(err_weekday),
            "improvement_pct": (1 - w / n) * 100 if n > 0 else 0.0}

            - 4 if it has its own vehicle

        penalty = learning.wait_penalty_km(wait) if config.ADAPTIVE_MATCHING else 0.0
        out.append({
            "ngo": ngo, "distance": round(dist, 1), "capacity_ok": capacity_ok, "vehicle": ngo.has_vehicle,
            "history": h, "p_accept": learning.acceptance_probability(h.accepted, h.total),
            "wait_min": wait, "penalty": round(penalty, 2), "score": round(score + penalty, 2),
        })
    out.sort(key=lambda r: r["score"])
    return out

    return [(r["ngo"], r["score"], r["distance"]) for r in rank_details(session, listing)]

def offer_next(session, listing: Listing) -> bool:
    """Offer the listing to the next-best NGO not yet asked. If none remain, open it to everyone."""
    already = set(session.scalars(select(Offer.ngo_id).where(Offer.listing_id == listing.id)))
    for ngo, _score, dist in rank_ngos(session, listing):
        if ngo.id in already:
            continue

        session.add(Offer(listing_id=listing.id, ngo_id=ngo.id, expires_at=expires))

            f"{dist} km away. Respond in the app within {format_duration(config.OFFER_TIMEOUT_SEC)}.",
        )
        return True

    listing.visible_to_all = True
    listing.opened_to_all_at = now()
    for ngo, _s, _d in rank_ngos(session, listing):
        notify(session, ngo.id, f"Open to all NGOs: {listing.quantity_meals} meals ({listing.title}).")
    return False

def heads_up(session, donor: User, expected_meals: int, when_text: str, limit: int = 3) -> list[str]:
    """Tell the nearest NGOs to expect surplus from this donor. Returns the NGO names alerted."""
    if donor.lat is None or expected_meals < 1:
        return []
    probe = SimpleNamespace(lat=donor.lat, lon=donor.lon, quantity_meals=expected_meals)
    names = []
    for ngo, _score, dist in rank_ngos(session, probe)[:limit]:
        notify(session, ngo.id,
               f"Heads-up: {donor.display_name} expects about {expected_meals} meals of surplus {when_text}, "
               f"{dist} km from you. No action is needed yet.")
        names.append(ngo.display_name)
    return names


def tick(session) -> dict:
    """Housekeeping: expire stale listings and escalate unanswered offers. Safe to call often."""
    n = now()
    expired_listings = escalated = 0

    stale = session.scalars(
        select(Listing).where(Listing.status == "open", (Listing.pickup_by <= n) | (Listing.safe_until <= n))
    ).all()
    for listing in stale:
        listing.status = "expired"
        listing.expired_at = n
        session.execute(
            update(Offer).where(Offer.listing_id == listing.id, Offer.status == "pending")
            .values(status="cancelled", resolved_at=n)
        )
        notify(session, listing.donor_id, f"Your listing '{listing.title}' expired without being collected.")
        expired_listings += 1

    timed_out = session.scalars(select(Offer).where(Offer.status == "pending", Offer.expires_at <= n)).all()
    for offer in timed_out:
        offer.status = "expired"
        offer.resolved_at = n
        listing = session.get(Listing, offer.listing_id)
        if listing and listing.status == "open":
            offer_next(session, listing)
            escalated += 1

    session.flush()
    return {"expired_listings": expired_listings, "escalated": escalated}

"""Notifications: always stored in-app; optionally also sent through Twilio (SMS or WhatsApp)."""
import os
import re

from core.db import Notification, User


def normalize_phone(raw: str, default_country: str = "+91") -> str:
    """'98450 12345' -> '+919845012345'. Numbers that already start with + are kept."""
    digits = re.sub(r"[^\d+]", "", raw or "")
    if digits.startswith("+"):
        return digits
    digits = digits.lstrip("0")
    return default_country + digits if digits else ""


def notify(session, user_id: int, message: str) -> None:
    session.add(Notification(user_id=user_id, message=message[:500]))
    user = session.get(User, user_id)
    if user and user.phone:
        _send_external(user.phone, message)


def _send_external(phone: str, message: str) -> tuple[bool, str]:
    """Returns (sent, detail). Quietly does nothing when Twilio is not configured."""
    sid, token, sender = (os.getenv(k) for k in ("TWILIO_ACCOUNT_SID", "TWILIO_AUTH_TOKEN", "TWILIO_FROM"))
    if not (sid and token and sender):
        return False, "Twilio is not configured, so only in-app alerts are used."
    to = normalize_phone(phone)
    if not to:
        return False, "No valid phone number."
    try:
        from twilio.rest import Client  # lazy import so Twilio stays optional

        target = f"whatsapp:{to}" if sender.startswith("whatsapp:") else to
        Client(sid, token).messages.create(body=message[:1500], from_=sender, to=target)
        return True, f"Sent to {to}."
    except Exception as exc:  # never let a failed alert break the main flow
        print(f"[notify] external send failed: {exc}")
        return False, f"Sending failed: {exc}"


def send_test(phone: str) -> tuple[bool, str]:
    return _send_external(phone, "SurplusLink test message. If you can read this, alerts are working.")

"""Coverage data: kitchens and venues from OpenStreetMap, cached to data/places.json.

Fetch once with internet access (`py scripts/fetch_places.py`), then the Coverage page works offline.
"""
import json
import urllib.parse
import urllib.request
from datetime import datetime

from core import config

OVERPASS_URLS = [
    "https://overpass-api.de/api/interpreter",
    "https://overpass.kumi.systems/api/interpreter",
    "https://overpass.private.coffee/api/interpreter",
]


def build_query(lat: float, lon: float, radius_km: float) -> str:
    r = int(radius_km * 1000)
    return (
        "[out:json][timeout:60];(\n"
        f'  nwr["amenity"~"^(restaurant|fast_food|canteen|food_court|cafe)$"](around:{r},{lat},{lon});\n'
        f'  nwr["tourism"~"^(hotel|guest_house|hostel)$"](around:{r},{lat},{lon});\n'
        f'  nwr["amenity"~"^(events_venue|community_centre|social_centre)$"](around:{r},{lat},{lon});\n'
        ");\nout center tags;"
    )


def classify(tags: dict) -> str:
    amenity, tourism = tags.get("amenity", ""), tags.get("tourism", "")
    if tourism in ("hotel", "guest_house", "hostel"):
        return "Hotel or lodging"
    if amenity in ("events_venue", "community_centre", "social_centre"):
        return "Event hall or venue"
    if amenity == "canteen":
        return "Canteen"
    return "Restaurant or cafe"


def parse_elements(data: dict) -> list[dict]:
    places = []
    for el in data.get("elements", []):
        tags = el.get("tags", {})
        lat = el.get("lat", el.get("center", {}).get("lat"))
        lon = el.get("lon", el.get("center", {}).get("lon"))
        if lat is None or lon is None:
            continue
        places.append({"name": tags.get("name", ""), "kind": classify(tags), "lat": float(lat), "lon": float(lon)})
    return places


def fetch(lat: float, lon: float, radius_km: float, urls=OVERPASS_URLS, timeout: int = 90) -> list[dict]:
    """Try each public Overpass server in turn. Raises RuntimeError if all fail."""
    body = urllib.parse.urlencode({"data": build_query(lat, lon, radius_km)}).encode()
    ua = f"SurplusLink/1.0 ({config.GEOCODE_CONTACT or 'student project'})"
    errors = []
    for url in urls:
        try:
            req = urllib.request.Request(url, data=body, headers={"User-Agent": ua})
            with urllib.request.urlopen(req, timeout=timeout) as resp:
                return parse_elements(json.load(resp))
        except Exception as exc:
            errors.append(f"{url}: {exc}")
    raise RuntimeError("All Overpass servers failed. " + " | ".join(errors))


def save(places: list[dict], city: str, radius_km: float) -> None:
    payload = {"fetched_at": datetime.now().isoformat(timespec="seconds"), "city": city, "radius_km": radius_km,
               "source": "OpenStreetMap contributors (ODbL)", "places": places}
    config.PLACES_FILE.write_text(json.dumps(payload), encoding="utf-8")


def load() -> dict | None:
    try:
        return json.loads(config.PLACES_FILE.read_text(encoding="utf-8"))
    except (OSError, ValueError):
        return None


def refresh() -> int:
    """Fetch for the configured coverage city and cache it. Returns the number of places."""
    lat, lon = config.CITY_PRESETS[config.COVERAGE_CITY]
    places = fetch(lat, lon, config.COVERAGE_RADIUS_KM)
    save(places, config.COVERAGE_CITY, config.COVERAGE_RADIUS_KM)
    return len(places)

    h, m = divmod(minutes, 60)
    return f"{h}h {m:02d}m left" if h else f"{m}m left"

    minutes = round(seconds / 60)
    if minutes < 120:
        return f"{minutes} minutes"
    return f"{round(minutes / 60)} hours"

"""Does learning from offer outcomes help? A transparent simulation.

Every NGO has hidden behaviour (how often it stays silent, how often it accepts when it does respond, how
quickly). Two policies face exactly the same random outcomes for exactly the same listings:

  fixed     asks the NGOs in order of distance, always
  learning  adds a penalty for NGOs that tend to waste time, learned only from what it has seen so far

The numbers are synthetic. They show whether the idea works under stated assumptions; they are not
evidence about real NGOs.
"""
from __future__ import annotations

import random
import statistics

from core import learning


def _outcomes(rng: random.Random, ngos: list[dict], timeout: float) -> list[tuple[str, float]]:
    """For one listing: what each NGO would do if asked. ('silent'|'declined'|'accepted', minutes taken)."""
    out = []
    for n in ngos:
        if rng.random() < n["p_silent"]:
            out.append(("silent", timeout))
            continue
        minutes = min(timeout, rng.expovariate(1 / n["resp_mean"]))
        out.append(("accepted" if rng.random() < n["p_accept"] else "declined", minutes))
    return out


def _run(policy: str, listings: list[dict], ngos: list[dict], timeout: float) -> list[dict]:
    history = {i: learning.NgoHistory() for i in range(len(ngos))}
    results = []
    for L in listings:
        scores = []
        for i in range(len(ngos)):
            s = L["dist"][i]
            if policy == "learning":
                s += learning.wait_penalty_km(learning.expected_wait_minutes(history[i], timeout))
            scores.append((s, i))
        waited, claimed, asked = 0.0, False, 0
        for _s, i in sorted(scores):
            kind, minutes = L["outcome"][i]
            asked += 1
            h = history[i]
            if kind == "accepted":
                waited += minutes
                h.accepted += 1
                h.accept_minutes += minutes
                claimed = True
                break
            waited += minutes
            h.declined += kind == "declined"
            h.expired += kind == "silent"
            h.fail_minutes += minutes
        results.append({"minutes": waited, "first_try": claimed and asked == 1, "claimed": claimed})
    return results


def _summary(results: list[dict]) -> dict:
    minutes = sorted(r["minutes"] for r in results)
    return {
        "mean_minutes": statistics.mean(minutes),
        "p90_minutes": minutes[int(0.9 * (len(minutes) - 1))],
        "first_offer_accepted": sum(r["first_try"] for r in results) / len(results),
        "fell_through": sum(not r["claimed"] for r in results) / len(results),
    }


def simulate(n_listings: int = 400, n_ngos: int = 6, timeout_min: float = 15.0, seed: int = 1) -> dict:
    rng = random.Random(seed)
    ngos = [{"p_silent": rng.uniform(0.05, 0.6), "p_accept": rng.uniform(0.5, 0.95), "resp_mean": rng.uniform(1.5, 8.0)}
            for _ in range(n_ngos)]
    listings = [{"dist": [rng.uniform(0.5, 12.0) for _ in range(n_ngos)], "outcome": _outcomes(rng, ngos, timeout_min)}
                for _ in range(n_listings)]
    fixed = _run("fixed", listings, ngos, timeout_min)
    learned = _run("learning", listings, ngos, timeout_min)
    block = max(n_listings // 4, 1)
    curve = [{"listings": f"{i + 1}-{i + block}",
              "fixed": statistics.mean(r["minutes"] for r in fixed[i:i + block]),
              "learning": statistics.mean(r["minutes"] for r in learned[i:i + block])}
             for i in range(0, n_listings - block + 1, block)]
    return {"fixed": _summary(fixed), "learning": _summary(learned), "curve": curve,
            "assumptions": {"listings": n_listings, "ngos": n_ngos, "timeout_min": timeout_min, "seed": seed}}

def countdown(until: datetime, at: datetime | None = None) -> str:
    secs = int((until - (at or now())).total_seconds())
    if secs <= 0:
        return "expired"
    minutes, seconds = divmod(secs, 60)
    hours, minutes = divmod(minutes, 60)
    return f"{hours}:{minutes:02d}:{seconds:02d}" if hours else f"{minutes}:{seconds:02d}"


def listing_html(l, *, distance_km: float | None = None, show_status: bool = False, show_donor: bool = True,
                 offer_expires: datetime | None = None, at: datetime | None = None) -> str:
    """One listing as a small specification sheet with the safe-time rail underneath."""
    at = at or now()
    total = max((l.safe_until - l.cooked_at).total_seconds(), 1.0)
    left_s = (l.safe_until - at).total_seconds()
    frac = min(max(left_s / total, 0.0), 1.0)
    tone = "ok" if frac >= 0.5 else "warn" if frac >= 0.25 else "bad"
    left_min = int(left_s // 60)
    remaining = "Safe time has ended" if left_min <= 0 else safety.format_minutes(left_min)

    sub = ([esc(l.donor.display_name)] if show_donor else []) + [esc(l.address)]
    spec = [("Meals", str(l.quantity_meals))]
    if distance_km is not None:
        spec.append(("Distance", f"{distance_km:.1f} km"))
    spec.append(("Planning limit", f"{l.safe_until:%H:%M}"))
    if l.status == "open":
        spec.append(("Collect by", f"{l.pickup_by:%H:%M}"))
    if offer_expires is not None:
        spec.append(("Offer expires in", countdown(offer_expires, at)))
    if l.temperature_c is not None:
        spec.append(("Packed at", f"{l.temperature_c:.0f} C"))
    spec_html = "".join(f"<div><dt>{esc(k)}</dt><dd>{esc(v)}</dd></div>" for k, v in spec)
    status = f'<div class="lstatus">{status_label(l.status)}</div>' if show_status else ""

    return (
        f'<div class="lrow"><div class="lhead"><div class="ltitle">{veg_mark(l.veg)}<span>{esc(l.title)}</span></div>{status}</div>'
        f'<div class="lsub">{", ".join(sub)}</div><dl class="lspec">{spec_html}</dl>'
        f'<div class="rail rail-{tone}" role="img" aria-label="{frac * 100:.0f} percent of planning time remaining">'
        f'<div style="width:{frac * 100:.0f}%"></div></div>'
        f'<div class="rail-text"><span>Time remaining against the planning limit</span><span>{remaining}</span></div></div>'
    )


def timeline_html(events: list[tuple[datetime, str]], pending=None, at: datetime | None = None) -> str:
    """Journey of a listing. `pending` is the open Offer, shown last with a live countdown."""
    at = at or now()

    def stamp(dt: datetime) -> str:
        return f"{dt:%H:%M:%S}" if dt.date() == at.date() else f"{dt:%d %b %H:%M}"

    items = [f'<li><span class="t">{stamp(t)}</span><span class="e">{esc(text)}</span></li>' for t, text in events]
    if pending is not None:
        items.append(f'<li class="live"><span class="t">Now</span><span class="e">Waiting for {esc(pending.ngo.display_name)}. '
                     f'Offer expires in {countdown(pending.expires_at, at)}</span></li>')
    return f'<ol class="tl">{"".join(items)}</ol>'

def location_input(prefix: str, label: str = "Address", default: tuple[float, float] | None = None):
    """Address search with a manual fallback. Returns (address_text, lat, lon)."""
    addr = st.text_input(label, key=f"{prefix}_addr", placeholder="Building, street, area, city")
    found_key = f"{prefix}_found"
    if st.button("Find on map", key=f"{prefix}_find"):
        res = geocode.geocode(addr)
        if res:
            st.session_state[found_key] = (res["lat"], res["lon"], res["name"], addr)
            st.session_state.pop(f"{prefix}_miss", None)
        else:
            st.session_state.pop(found_key, None)
            st.session_state[f"{prefix}_miss"] = True
    found = st.session_state.get(found_key)
    if found and found[3] != addr:
        st.caption("The address has changed since the last search. Press Find on map again.")
        found = None
    missed = st.session_state.get(f"{prefix}_miss", False)
    if missed and not found:
        st.warning("No match found for that address. Add the area and city, or set the location below.")

    cities = list(config.CITY_PRESETS)
    with st.expander("Choose a city or enter coordinates instead", expanded=bool(missed and not found)):
        city = st.selectbox("City", cities, index=cities.index(config.COVERAGE_CITY), key=f"{prefix}_city")
        start = default if default else config.CITY_PRESETS[city]
        c1, c2 = st.columns(2)
        lat_m = c1.number_input("Latitude", value=float(start[0]), format="%.5f", key=f"{prefix}_lat_{city}")
        lon_m = c2.number_input("Longitude", value=float(start[1]), format="%.5f", key=f"{prefix}_lon_{city}")

    if found:
        lat, lon = found[0], found[1]
        st.caption(f"Matched to: {found[2]}")
        st.map(pd.DataFrame({"lat": [lat], "lon": [lon]}), latitude="lat", longitude="lon", size=40, zoom=14, height=220)
    else:
        lat, lon = lat_m, lon_m
        st.caption(f"No address match is in use. The position is {city} unless you edited the coordinates, "
                   "so distances will be approximate.")
    st.caption(geocode.ATTRIBUTION)
    return addr, lat, lon

# SurplusLink

Hotels, caterers and event kitchens list surplus cooked food. The best-matched NGO is offered it first;
if it does not respond in time, the offer moves to the next NGO. A volunteer or the NGO collects it, and every
delivery produces a record laid out in the format of FSSAI Schedule II.

## What it does

- **Matching with a visible cascade.** NGOs are ranked by distance, capacity and vehicle. Each listing has a live
  timeline showing every offer, who did not respond, and who claimed it.
- **Address search.** Donors and NGOs find their address on a map (OpenStreetMap Nominatim) instead of picking a city.
- **Surplus food record.** Donor licence and NGO registration numbers, batch reference, food temperature, quantity
  distributed and area distributed are captured, and a PDF can be downloaded for each delivery.
- **NGO approval checks.** NGOs give an FSSAI registration number and NGO Darpan ID; an administrator reviews them before
  approving. Darpan IDs are self-declared, so they are something to check, not proof.
- **Alerts.** In-app notifications always; SMS or WhatsApp through Twilio if configured. Donors can send a heads-up to
  the nearest NGOs based on tomorrow's forecast.
- **Coverage.** A map of kitchens and venues from OpenStreetMap next to the NGOs registered on the platform.
- **Impact.** Meals delivered, estimated food and emissions avoided, and a next-day forecast with a range.

## Where the intelligence is, and what it is not

- **Matching that learns.** From every past offer (accepted, declined, or never answered, and how long it took) the app
  estimates how likely each NGO is to accept and how long it typically wastes. That becomes a waiting penalty added to the
  distance rule. Estimates are smoothed so a few offers do not decide anything. Each donor can see the working under
  "Why NGOs are asked in this order", and Administration shows what has been learned. This is simple statistics (Bayesian
  smoothing), not a neural network.
- **A simulation of whether it helps.** On the Impact page, 400 synthetic listings are replayed under two policies,
  distance only and distance plus learning, against identical hidden NGO behaviour. In five runs with different seeds the
  learning policy waited less on average each time (by roughly 3 to 19 percent). The behaviour is invented, so this
  shows the idea works under stated assumptions, not that real NGOs behave this way.
- **A forecast that tests itself.** Tomorrow's surplus is the same-weekday average with days of no surplus counted as zero,
  with a range. The Impact page back-tests it over the last two weeks against assuming every day is average. A regression on
  weekday and trend was tried on demo-like data and did not beat the simple weekday average, so it is not used.
- **An optional language-model assistant.** If `ANTHROPIC_API_KEY` is set, a donor can describe the food in a sentence and the
  form is pre-filled. Every value is validated, the donor checks it before posting, and nothing is posted automatically.
  The text typed is sent to the Anthropic API.
- **Not AI:** the address search (OpenStreetMap), the distance formula, the safe-time table and the impact estimates are
  ordinary rules and lookups. The demo history is synthetic.

## Limits to be honest about

- The planning limits in `core/safety.py` are defaults used to schedule pickups. They are not regulatory limits. Food
  safety remains the responsibility of the donor and the distribution organisation.
- The record is laid out like Schedule II but is not a legal certificate. Check the current regulation text and keep a
  signed copy.
- Login is demo-grade: no email verification, password reset or attempt limits.
- The forecast is a same-weekday average. It needs real history to mean anything.
- The coverage counts come from OpenStreetMap, which is volunteer-edited and incomplete.

"""Optional background worker: expires listings and escalates unanswered offers every minute.

The Streamlit app already runs the same housekeeping while someone has it open. Run this too
(`python worker.py` in a second terminal) so escalation continues when nobody is browsing.
"""
from apscheduler.schedulers.blocking import BlockingScheduler

from core import config, matching
from core.db import init_db, session_scope


def run_tick() -> None:
    with session_scope() as s:
        result = matching.tick(s)
    if any(result.values()):
        print("tick:", result)


if __name__ == "__main__":
    init_db()
    scheduler = BlockingScheduler()
    scheduler.add_job(run_tick, "interval", seconds=config.TICK_SECONDS, max_instances=1)
    print("Worker running. Ctrl+C to stop.")
    try:
        scheduler.start()
    except (KeyboardInterrupt, SystemExit):
        pass
